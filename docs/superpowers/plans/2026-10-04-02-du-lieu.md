### A. Độ chính xác vị trí cho bộ IRD (Task 2)

```python
def ird_precision(spatial_precision, geocoding_method) -> str:
    method = str(geocoding_method).strip().lower().replace("_", " ")
    if method.startswith(("area", "ward")):
        return "area"
    precision = str(spatial_precision).strip().lower()
    if precision in {"high", "hugh", "hgh"}:
        return "high"
    return "medium" if precision == "medium" else "low"
```

```python
from shapely.ops import substring


def cut_pieces(line, max_len: float) -> list:
    n = max(1, int(np.ceil(line.length / max_len)))
    step = line.length / n
    return [substring(line, i * step, (i + 1) * step) for i in range(n)]


def _tagged(value) -> bool:
    value = first(value)
    return value is not None and not pd.isna(value) and str(value).lower() != "no"
```

### D. Gán ghi nhận vào đoạn (Task 7b)

Thay `match_units` bằng hàm dưới đây. Lệnh `python -m floodrisk.data.observations map hcm` ghi lại `observations.parquet` (cột `unit_id` là đoạn neo) và ghi `obs_units.parquet`.

```python
LIMIT_M = {"medium": 500.0, "low": 1500.0}


def _chain(u, keys, anchor: int, point, limit_m: float, link_m: float) -> list[int]:
    key = keys.iloc[anchor]
    near = [i for i in u.sindex.query(point.buffer(limit_m))
            if keys.iloc[i] == key and u.geometry.iloc[i].distance(point) <= limit_m]
    pool, seen, queue = set(near) | {anchor}, {anchor}, [anchor]
    while queue:
        current = queue.pop()
        for other in pool - seen:
            if u.geometry.iloc[current].distance(u.geometry.iloc[other]) <= link_m:
                seen.add(other)
                queue.append(other)
    return sorted(seen)


def map_observations(obs, units, near_m=100.0, name_m=300.0, link_m=30.0):
    crs = units.estimate_utm_crs()
    u = units[["unit_id", "name", "length_m", "geometry"]].to_crs(crs).reset_index(drop=True)
    keys = u["name"].map(norm_name)
    points = gpd.GeoSeries(gpd.points_from_xy(obs["lon"], obs["lat"]), crs=4326).to_crs(crs)
    anchors, rows, by_name = [], [], 0
    for obs_id, point, street, precision in zip(obs["obs_id"], points, obs["street_name"], obs["loc_precision"]):
        anchor = None
        if precision != "area":
            candidates = u.sindex.query(point.buffer(name_m))
            if len(candidates):
                dist = u.geometry.iloc[candidates].distance(point)
                key = norm_name(street) if isinstance(street, str) else ""
                same = dist[keys.iloc[candidates] == key] if key else dist.iloc[0:0]
                pick = same if len(same) else dist[dist <= near_m]
                if len(pick):
                    anchor = int(pick.idxmin())
                    by_name += int(len(same) > 0)
        anchors.append(None if anchor is None else u.at[anchor, "unit_id"])
        if anchor is None:
            continue
        if precision == "high":
            members = [anchor] if u.geometry.iloc[anchor].distance(point) <= near_m else []
        else:
            members = _chain(u, keys, anchor, point, LIMIT_M[precision], link_m)
        spread = float(u["length_m"].iloc[members].sum())
        rows += [{"obs_id": obs_id, "unit_id": u.at[m, "unit_id"], "exact": precision == "high", "spread_m": spread}
                 for m in members]
    stats = {"total": len(obs), "anchored": sum(a is not None for a in anchors), "by_name": by_name}
    table = pd.DataFrame(rows, columns=["obs_id", "unit_id", "exact", "spread_m"])
    return pd.Series(anchors, index=obs.index, dtype=object), table, stats
```

Lệnh `map` phải in `stats` và 20 tên đường không được neo nhiều nhất (QĐKT mục 4.2).

Kiểm thử, trên một lưới đoạn tổng hợp gồm một con đường thẳng dài 2 km cắt thành mười đoạn 200 m và một con đường khác trùng tên nằm cách 1 km:
- Ghi nhận `high` cách đoạn thứ ba 20 m: gán đúng một đoạn, `exact` đúng, `spread_m` bằng 200.
- Ghi nhận `high` cách mọi đoạn trên 100 m: không gán.
- Ghi nhận `medium` có tên đường, nằm trên đoạn thứ năm: gán các đoạn trong 500 m, tức khoảng năm đoạn liền nhau; `exact` sai.
- Ghi nhận `low`: gán các đoạn trong 1.500 m của **cùng** con đường, và không gán đoạn nào của con đường trùng tên ở xa.
- Ghi nhận `area`: không gán, đoạn neo rỗng.
- Ghi nhận không có tên đường, mức `medium`: neo vào đoạn gần nhất trong 100 m rồi lan theo tên của đoạn đó.

### G. Mạng đường cho thuật toán tìm đường (Task 5)

Goong chỉ trả 1 hoặc 2 lộ trình (QĐKT mục 7.3), nên kỹ sư phần mềm viết thuật toán tìm đường riêng và cần mạng đường từ chính đồ thị OSMnx đã tải để tạo đoạn. Ghi ba file theo hợp đồng ở QĐKT mục 5:

- **`graph_nodes.parquet`:** từ `ox.graph_to_gdfs(graph, edges=False)`; `node_id` là mã nút OpenStreetMap, `lon` và `lat` là tọa độ.
- **`graph_edges.parquet`** (GeoParquet, EPSG:4326): từ `ox.graph_to_gdfs(graph, nodes=False).reset_index()`. Mỗi dòng là một chiều đi được, kể cả cạnh không tên. Khi có nhiều cạnh song song cùng `u`, `v` thì chỉ giữ cạnh ngắn nhất. `edge_id` là số thứ tự dòng sau khi lọc; `length_m` lấy từ cột `length`; `highway` lấy phần tử đầu nếu là danh sách.
- **`edge_units.parquet`:** `split_into_units` nhận thêm cột `edge_id` trên bảng cạnh và trả thêm bảng mảnh `(edge_id, unit_id, length_m)`; cộng dồn chiều dài theo cặp (cạnh, đoạn). Chữ ký mới: `split_into_units(edges, city, cell_m, parent_m) -> (units, pieces)`.

Kiểm thử:
- Một đường hai chiều cho hai cạnh ngược nhau, cả hai trỏ tới cùng các đoạn.
- Cạnh không tên có trong `graph_edges` nhưng không có dòng nào trong `edge_units`.
- Với cạnh có tên, tổng `length_m` trong `edge_units` bằng chiều dài cạnh, sai lệch dưới 1 m.
- Không có hai dòng nào trong `graph_edges` trùng cả `u` lẫn `v`.
- Ba bảng qua được `contracts.GRAPH_NODES`, `GRAPH_EDGES`, `EDGE_UNITS`.

### H. Định vị 34 tuyến chính thức bằng cách khớp tên (Task 2)

File `data/raw/hcm_official/streets.csv` có các cột `name`, `ward`, `cause`, và hai cột tùy chọn `lon`, `lat`.

Hàm `build()` cũ tìm file `streets_geocoded.csv`; bỏ đoạn đó đi. Thay bằng lệnh `python -m floodrisk.data.observations official hcm`, chạy sau khi có `units.parquet` và trước lệnh `map`: nó gọi `load_official`, thay các dòng `hcm_official` cũ trong `observations.parquet` bằng kết quả mới.

`load_official(csv_path, units)`:

1. Lấy các đoạn có tên chuẩn hóa (`norm_name`) trùng với tên trong file.
2. Nhóm chúng thành các cụm liền nhau (hai đoạn cách nhau không quá 30 m là liền nhau).
3. Nếu chỉ có một cụm, hoặc dòng đó có sẵn `lon`, `lat`: tạo một ghi nhận `low` đặt tại điểm giữa của cụm (hoặc tại tọa độ đã cho).
4. Nếu có nhiều cụm cách xa nhau và không có tọa độ: in tên đường, phường, và tâm của từng cụm, rồi bỏ qua dòng đó. Người dữ liệu tra phường trên bản đồ, điền `lon`, `lat` vào file, và chạy lại.
5. Nếu không có đoạn nào trùng tên: in ra để soát lỗi chính tả.

Kiểm thử: tên có một cụm cho đúng một ghi nhận; tên có hai cụm cách xa và không có tọa độ thì bị bỏ qua và được in ra; có tọa độ thì ghi nhận nằm ở tọa độ đó; tên không tồn tại không làm hỏng hàm.

```python
from __future__ import annotations

import json
import logging
import os
import time
from datetime import datetime, timezone
from pathlib import Path

import requests

from floodrisk.config import CITIES, data_dir

log = logging.getLogger("floodrisk.loggers")

FIXED = {
    "danang_reports": "https://muangap-api.danang.gov.vn/v1/flood/reports",
    "danang_water_stations": "https://muangap-api.danang.gov.vn/v2/client/water_station/list_all",
    "danang_rain_stations": "https://muangap-api.danang.gov.vn/v1/stations?page=0&size=-1",
    "hanoi_flood": "https://thoatnuochanoi.vn/ungngap/api/flood/getflood",
}


def sources() -> dict[str, str]:
    out = dict(FIXED)
    for city in CITIES.values():
        lon, lat = city.center
        out[f"openmeteo_{city.key}"] = (
            "https://api.open-meteo.com/v1/forecast"
            f"?latitude={lat}&longitude={lon}&hourly=precipitation"
            "&past_days=1&forecast_days=2&timezone=Asia%2FHo_Chi_Minh"
        )
    return out


def snapshot(name: str, url: str, out_dir: Path, get=requests.get, now: datetime | None = None) -> Path | None:
    now = now or datetime.now(timezone.utc)
    contact = os.environ.get("FLOODRISK_CONTACT", "no-contact")
    try:
        resp = get(url, headers={"User-Agent": f"floodrisk-hackathon ({contact})"}, timeout=30)
        resp.raise_for_status()
        payload = resp.json()
    except Exception as exc:  # một nguồn lỗi không được làm dừng vòng ghi
        log.warning("Không lấy được %s: %s", name, exc)
        return None
    path = Path(out_dir) / name / f"{now:%Y-%m-%d}.jsonl"
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps({"fetched_at": now.isoformat(), "payload": payload}, ensure_ascii=False) + "\n")
    return path


def run_forever(interval_s: int = 900, out_dir: Path | None = None) -> None:
    out_dir = out_dir or data_dir() / "raw" / "live"
    while True:
        for name, url in sources().items():
            snapshot(name, url, out_dir)
        time.sleep(interval_s)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    run_forever()
```

```python
from __future__ import annotations

import json
import os
import re
import sys
import unicodedata
from pathlib import Path

import geopandas as gpd
import numpy as np
import pandas as pd
import requests

from floodrisk import contracts
from floodrisk.config import TZ, data_dir, observations_path

COLUMNS = list(contracts.OBSERVATIONS)
IRD_CAUSE = {"rain": "rain", "high tide": "tide", "combined": "combined"}
IRD_CLASS = {"low": 1.0, "moderate": 2.0, "heavy": 3.0, "severe": 3.0}
GEOCODE_URL = "https://rsapi.goong.io/v2/geocode"


def depth_class(cm) -> float:
    if cm is None or pd.isna(cm):
        return np.nan
    if cm < 10:
        return 1.0
    return 2.0 if cm <= 30 else 3.0


def _norm_cols(df: pd.DataFrame) -> pd.DataFrame:
    def key(column) -> str:
        ascii_ = unicodedata.normalize("NFD", str(column).lower()).encode("ascii", "ignore").decode()
        return re.sub(r"[^a-z0-9]+", "_", ascii_).strip("_")

    return df.rename(columns={c: key(c) for c in df.columns})


def _col(df: pd.DataFrame, *candidates: str) -> pd.Series:
    for c in candidates:
        if c in df.columns:
            return df[c]
    raise KeyError(f"Không tìm thấy cột nào trong {candidates}; các cột hiện có: {list(df.columns)}")


def _parse_date(s: pd.Series) -> pd.Series:
    if pd.api.types.is_datetime64_any_dtype(s):
        return s
    return pd.to_datetime(s.astype(str).str.strip(), format="%d/%m/%Y", errors="coerce")


def _finish(df: pd.DataFrame) -> pd.DataFrame:
    df = df.reindex(columns=COLUMNS)
    df["date"] = pd.to_datetime(df["date"])
    for c in ("year", "hour", "lon", "lat", "depth_cm", "depth_class"):
        df[c] = pd.to_numeric(df[c], errors="coerce").astype(float)
    df["provenance"] = df["provenance"].astype(int)
    df["flooded"] = df["flooded"].astype(bool)
    return df.reset_index(drop=True)


def load_ird(gpkg_path) -> pd.DataFrame:
    g = gpd.read_file(gpkg_path).to_crs(4326)
    pts = g.geometry.representative_point()
    df = _norm_cols(pd.DataFrame(g.drop(columns=g.geometry.name)))
    obs_id = _col(df, "obs_id")
    date = _parse_date(_col(df, "date"))
    year = pd.to_numeric(_col(df, "year"), errors="coerce").fillna(date.dt.year)
    depth = pd.to_numeric(_col(df, "water_height_max_cm"), errors="coerce").fillna(
        pd.to_numeric(_col(df, "water_height_min_cm"), errors="coerce")
    )
    named_class = _col(df, "depth_class").astype(str).str.strip().str.lower().map(IRD_CLASS)
    cause = _col(df, "cause").astype(str).str.strip().str.lower().map(IRD_CAUSE).fillna("unknown")
    method = _col(df, "geocoding_method").astype(str).str.lower()
    return _finish(pd.DataFrame({
        "obs_id": "ird-" + obs_id.astype(str),
        "city": "hcm", "source": "ird", "provenance": 1,
        "date": date, "year": year, "hour": np.nan,
        "lon": pts.x.to_numpy(), "lat": pts.y.to_numpy(),
        "loc_precision": np.where(method.str.contains("street"), "street", "point"),
        "flooded": True,
        "depth_cm": depth, "depth_class": depth.map(depth_class).fillna(named_class),
        "cause": cause, "street_name": _col(df, "location_name").astype(str),
        "unit_id": None, "evidence_url": "https://doi.org/10.23708/8Y16HU",
    }))


def load_danang(json_paths) -> pd.DataFrame:
    records: dict[str, dict] = {}
    for path in json_paths:
        payload = json.loads(Path(path).read_text(encoding="utf-8"))
        for rec in payload["data"] if isinstance(payload, dict) else payload:
            records[rec["_id"]] = rec
    rows = []
    for rec in records.values():
        coords = (rec.get("location") or {}).get("coordinates") or [None, None]
        start = (rec.get("flood_time") or {}).get("start_time")
        if coords[0] is None or start is None:
            continue
        ts = pd.Timestamp(start, unit="s", tz="UTC").tz_convert(TZ).tz_localize(None)
        level = rec.get("water_level")
```

```python
from __future__ import annotations

import numpy as np
import pandas as pd

from floodrisk.config import CITIES, RAIN_CELL_DEG


def rain_cells(city: str) -> pd.DataFrame:
    west, south, east, north = CITIES[city].bbox
    lons = np.arange(west + RAIN_CELL_DEG / 2, east, RAIN_CELL_DEG)
    lats = np.arange(south + RAIN_CELL_DEG / 2, north, RAIN_CELL_DEG)
    rows = [
        {"city": city, "cell_id": i, "lon": round(float(lon), 4), "lat": round(float(lat), 4)}
        for i, (lat, lon) in enumerate((lat, lon) for lat in lats for lon in lons)
    ]
    return pd.DataFrame(rows)


def nearest_cell(lons: np.ndarray, lats: np.ndarray, cells: pd.DataFrame) -> np.ndarray:
    d = (np.asarray(lons)[:, None] - cells["lon"].to_numpy()[None, :]) ** 2 + (
        np.asarray(lats)[:, None] - cells["lat"].to_numpy()[None, :]
    ) ** 2
    return cells["cell_id"].to_numpy()[d.argmin(axis=1)]
```

```python
from __future__ import annotations

import json
import sys
import time
from pathlib import Path

import pandas as pd
import requests

from floodrisk import contracts
from floodrisk.config import RAIN_HISTORY_START, TZ, data_dir, rain_cells_path, rain_daily_path
from floodrisk.data.cells import rain_cells

ARCHIVE = "https://archive-api.open-meteo.com/v1/archive"
FORECAST = "https://api.open-meteo.com/v1/forecast"


def _hourly_frame(cell_id: int, payload: dict) -> pd.DataFrame:
    hourly = payload["hourly"]
    return pd.DataFrame({
        "cell_id": int(cell_id),
        "time": pd.to_datetime(hourly["time"]),
        "precip_mm": pd.to_numeric(pd.Series(hourly["precipitation"], dtype="object"), errors="coerce").astype(float),
    })


def fetch_history(city: str, cells: pd.DataFrame, start, end, cache_dir, get=requests.get, pause_s: float = 1.0) -> pd.DataFrame:
    start, end = pd.Timestamp(start), pd.Timestamp(end)
    frames = []
    for cell in cells.itertuples():
        for year in range(start.year, end.year + 1):
            a = max(start, pd.Timestamp(year=year, month=1, day=1))
            b = min(end, pd.Timestamp(year=year, month=12, day=31))
            path = Path(cache_dir) / city / f"{cell.cell_id}_{a:%Y%m%d}_{b:%Y%m%d}.json"
            if not path.exists():
                resp = get(ARCHIVE, params={
                    "latitude": cell.lat, "longitude": cell.lon,
                    "start_date": f"{a:%Y-%m-%d}", "end_date": f"{b:%Y-%m-%d}",
                    "hourly": "precipitation", "timezone": TZ,
                }, timeout=120)
                resp.raise_for_status()
                path.parent.mkdir(parents=True, exist_ok=True)
                path.write_text(resp.text, encoding="utf-8")
                time.sleep(pause_s)
            frames.append(_hourly_frame(cell.cell_id, json.loads(path.read_text(encoding="utf-8"))))
    return pd.concat(frames, ignore_index=True)


def daily_features(hourly: pd.DataFrame) -> pd.DataFrame:
    h = hourly.sort_values(["cell_id", "time"]).copy()
    h["precip_mm"] = h["precip_mm"].fillna(0.0)
    h["r3"] = h.groupby("cell_id")["precip_mm"].transform(lambda s: s.rolling(3, min_periods=1).sum())
    h["date"] = h["time"].dt.normalize()
    return h.groupby(["cell_id", "date"], as_index=False).agg(r3max=("r3", "max"), r24=("precip_mm", "sum"))


def fetch_forecast(city: str, cells: pd.DataFrame, get=requests.get) -> pd.DataFrame:
    resp = get(FORECAST, params={
        "latitude": ",".join(f"{v:.4f}" for v in cells["lat"]),
        "longitude": ",".join(f"{v:.4f}" for v in cells["lon"]),
        "hourly": "precipitation", "past_days": 1, "forecast_days": 2, "timezone": TZ,
    }, timeout=60)
    resp.raise_for_status()
    payload = resp.json()
    items = payload if isinstance(payload, list) else [payload]
    if len(items) != len(cells):
        raise ValueError(f"{city}: Open-Meteo trả {len(items)} điểm, cần {len(cells)}")
    return pd.concat([_hourly_frame(cid, item) for cid, item in zip(cells["cell_id"], items)], ignore_index=True)


def build(city: str) -> Path:
    cells = rain_cells(city)
    contracts.validate(cells, contracts.RAIN_CELLS, "rain_cells")
    rain_cells_path(city).parent.mkdir(parents=True, exist_ok=True)
    cells.to_parquet(rain_cells_path(city))
    end = pd.Timestamp.now(tz=TZ).tz_localize(None).normalize() - pd.Timedelta(days=6)
    hourly = fetch_history(city, cells, RAIN_HISTORY_START, end, data_dir() / "raw" / "rain")
    daily = daily_features(hourly)
    contracts.validate(daily, contracts.RAIN_DAILY, "rain_daily")
    daily.to_parquet(rain_daily_path(city))
    return rain_daily_path(city)


if __name__ == "__main__":
    print(build(sys.argv[1]))
```

```python
from __future__ import annotations

import joblib
import numpy as np
import pandas as pd

from floodrisk import contracts
from floodrisk.config import RAIN_HISTORY_START, TZ, data_dir, tide_coef_path, tide_daily_path

EPOCH = "1970-01-01"
UTC_OFFSET = pd.Timedelta(hours=7)  # Việt Nam không đổi giờ theo mùa


def load_uhslc(path) -> pd.Series:
    df = pd.read_csv(path, header=None, names=["year", "month", "day", "hour", "mm"])
    times = pd.to_datetime(df[["year", "month", "day", "hour"]])
    metres = df["mm"].where(df["mm"] > -30000) / 1000.0
    s = pd.Series(metres.to_numpy(), index=pd.DatetimeIndex(times)).dropna()
    return s - s.mean()


def _days(times) -> np.ndarray:
    return (pd.DatetimeIndex(times).to_numpy() - np.datetime64(EPOCH)) / np.timedelta64(1, "D")


def fit(series: pd.Series, lat: float = 10.34):
    from utide import solve

    return solve(_days(series.index), series.to_numpy(), lat=lat, epoch=EPOCH,
                 method="ols", conf_int="none", trend=False, verbose=False)


def predict_utc(coef, times_utc) -> np.ndarray:
    from utide import reconstruct

    out = reconstruct(_days(times_utc), coef, epoch=EPOCH, verbose=False)
    return np.asarray(out.h) - float(coef["mean"])


def predict_local(coef, times_local) -> np.ndarray:
    return predict_utc(coef, pd.DatetimeIndex(times_local) - UTC_OFFSET)


def daily_max(coef, start, end) -> pd.DataFrame:
    times = pd.date_range(pd.Timestamp(start), pd.Timestamp(end) + pd.Timedelta(hours=23), freq="h")
    heights = pd.Series(predict_local(coef, times), index=times)
    out = heights.groupby(heights.index.normalize()).max()
    return pd.DataFrame({"date": out.index, "tide_max": out.to_numpy()})


def window_max(coef, at_local: pd.Timestamp, hours: int = 4) -> float:
    times = pd.date_range(at_local - pd.Timedelta(hours=hours), at_local, freq="h")
    return float(np.max(predict_local(coef, times)))


def build() -> None:
    series = load_uhslc(data_dir() / "raw" / "tide" / "h383.csv")
    coef = fit(series)
    tide_coef_path().parent.mkdir(parents=True, exist_ok=True)
    joblib.dump(coef, tide_coef_path())
    end = pd.Timestamp.now(tz=TZ).tz_localize(None).normalize() + pd.Timedelta(days=30)
    daily = daily_max(coef, RAIN_HISTORY_START, end)
```

```python
from __future__ import annotations

import hashlib
import unicodedata

import geopandas as gpd
import numpy as np
from shapely.ops import unary_union

from floodrisk.config import CITIES, GRID_CELL_M

ROAD_RANK = {"motorway": 6, "trunk": 5, "primary": 4, "secondary": 3, "tertiary": 2}
NAME_PREFIXES = ("duong ", "pho ")


def first(value):
    if isinstance(value, (list, tuple)):
        return value[0] if len(value) else None
    return value


def norm_name(name: str) -> str:
    s = unicodedata.normalize("NFD", str(name).lower().replace("đ", "d"))
    s = " ".join("".join(ch for ch in s if unicodedata.category(ch) != "Mn").split())
    for prefix in NAME_PREFIXES:
        if s.startswith(prefix):
            s = s[len(prefix):]
    return s


def road_rank(highway: str) -> int:
    return ROAD_RANK.get(str(highway).replace("_link", ""), 1)


def split_into_units(edges: gpd.GeoDataFrame, city: str, cell_m: int = GRID_CELL_M) -> gpd.GeoDataFrame:
    df = edges.copy()
    df["name"] = df["name"].map(first)
    df["highway"] = df["highway"].map(first)
    df = df[df["name"].notna() & df["highway"].notna()].copy()
    df["key"] = df["name"].map(norm_name)
    df = df[df["key"] != ""].copy()
    mid = df.geometry.interpolate(0.5, normalized=True)
    df["cx"] = np.floor(mid.x / cell_m).astype(int)
    df["cy"] = np.floor(mid.y / cell_m).astype(int)
    df["len"] = df.geometry.length

    rows = []
    for (key, cx, cy), group in df.groupby(["key", "cx", "cy"], sort=True):
        road_class = group.groupby("highway")["len"].sum().idxmax()
        geom = unary_union(list(group.geometry))
        digest = hashlib.sha1(f"{key}|{cx}|{cy}".encode()).hexdigest()[:10]
        rows.append({
            "unit_id": f"{city}-{digest}", "city": city, "name": group["name"].iloc[0],
            "road_class": road_class, "road_rank": road_rank(road_class),
            "length_m": float(geom.length), "geometry": geom,
        })
    return gpd.GeoDataFrame(rows, geometry="geometry", crs=edges.crs)


def download_edges(city: str) -> gpd.GeoDataFrame:
    import osmnx as ox

    ox.settings.requests_timeout = 300
    graph = ox.graph_from_bbox(CITIES[city].bbox, network_type="drive", simplify=True)
    return ox.graph_to_gdfs(graph, nodes=False).reset_index()[["name", "highway", "geometry"]]


def build_units(city: str) -> gpd.GeoDataFrame:
    edges = download_edges(city)
    projected = edges.to_crs(edges.estimate_utm_crs())
    return split_into_units(projected, city).to_crs(4326)
```

```python
from __future__ import annotations

from pathlib import Path

import numpy as np
import pandas as pd
import rasterio
from pyproj import Transformer
from rasterio.merge import merge
from rasterio.transform import rowcol
from rasterio.windows import from_bounds
from scipy.ndimage import uniform_filter

DEG_M = 111_320.0
WORLDCOVER = "https://esa-worldcover.s3.eu-central-1.amazonaws.com/v200/2021/map/ESA_WorldCover_10m_2021_v200_{tile}_Map.tif"
BUILT_UP = 50


def load_dem(tile_dir, bbox) -> tuple[np.ndarray, rasterio.Affine]:
    paths = sorted(Path(tile_dir).glob("*.tif"))
    if not paths:
        raise FileNotFoundError(f"Không có file .tif nào trong {tile_dir}")
    sources = [rasterio.open(p) for p in paths]
    try:
        west, south, east, north = bbox
        hits = [s for s in sources
                if not (s.bounds.right <= west or s.bounds.left >= east or s.bounds.top <= south or s.bounds.bottom >= north)]
        if not hits:
            raise ValueError(f"Không có ô DEM nào phủ khung {bbox}")
        data, transform = merge(hits, bounds=bbox, nodata=np.nan, dtype="float32")
    finally:
        for s in sources:
            s.close()
    dem = data[0]
    dem[dem < -100] = np.nan
    return dem, transform


def focal_mean(a: np.ndarray, size: int) -> np.ndarray:
    valid = np.isfinite(a)
    total = uniform_filter(np.where(valid, a, 0.0).astype("float64"), size=size, mode="nearest")
    count = uniform_filter(valid.astype("float64"), size=size, mode="nearest")
    with np.errstate(invalid="ignore", divide="ignore"):
        out = total / count
    out[count < 1e-9] = np.nan
    return out


def tpi(dem: np.ndarray, size: int) -> np.ndarray:
    return dem - focal_mean(dem, size)


def sample(arr: np.ndarray, transform, lons, lats) -> np.ndarray:
    rows, cols = rowcol(transform, np.asarray(lons), np.asarray(lats))
    rows, cols = np.asarray(rows), np.asarray(cols)
    inside = (rows >= 0) & (rows < arr.shape[0]) & (cols >= 0) & (cols < arr.shape[1])
    out = np.full(len(rows), np.nan, dtype="float64")
    out[inside] = arr[rows[inside], cols[inside]]
    return out


def line_points(geom, step_m: float) -> list:
    lines = geom.geoms if geom.geom_type.startswith("Multi") else [geom]
    points = []
    for line in lines:
        n = max(1, int(line.length // step_m))
        points.extend(line.interpolate(i / n, normalized=True) for i in range(n + 1))
    return points


def _odd(x: float) -> int:
    n = max(3, int(round(x)))
    return n if n % 2 else n + 1


def _agg(fn, values: np.ndarray) -> float:
    return float(fn(values)) if np.isfinite(values).any() else float("nan")


def _unit_lonlat(units, step_m: float):
    crs = units.estimate_utm_crs()
    back = Transformer.from_crs(crs, 4326, always_xy=True)
    for unit_id, geom in zip(units["unit_id"], units.geometry.to_crs(crs)):
        pts = line_points(geom, step_m)
        lons, lats = back.transform([p.x for p in pts], [p.y for p in pts])
        yield unit_id, np.asarray(lons), np.asarray(lats)


def terrain_features(units, dem: np.ndarray, transform, step_m: float = 30.0) -> pd.DataFrame:
    px_m = abs(transform.a) * DEG_M
    t300 = tpi(dem, _odd(600 / px_m))
    t1000 = tpi(dem, _odd(2000 / px_m))
    rows = []
    for unit_id, lons, lats in _unit_lonlat(units, step_m):
        z = sample(dem, transform, lons, lats)
        rows.append({
            "unit_id": unit_id,
            "elev_min": _agg(np.nanmin, z),
            "elev_mean": _agg(np.nanmean, z),
            "tpi_300": _agg(np.nanmin, sample(t300, transform, lons, lats)),
            "tpi_1000": _agg(np.nanmin, sample(t1000, transform, lons, lats)),
        })
    return pd.DataFrame(rows)


def worldcover_url(lon: float, lat: float) -> str:
    lat0, lon0 = int(np.floor(lat / 3) * 3), int(np.floor(lon / 3) * 3)
    tile = f"{'N' if lat0 >= 0 else 'S'}{abs(lat0):02d}{'E' if lon0 >= 0 else 'W'}{abs(lon0):03d}"
    return WORLDCOVER.format(tile=tile)


def load_built(bbox) -> tuple[np.ndarray, rasterio.Affine]:
    with rasterio.open(worldcover_url(bbox[0], bbox[1])) as src:
        window = from_bounds(*bbox, transform=src.transform).round_offsets().round_lengths()
        data = src.read(1, window=window)
        transform = src.window_transform(window)
    return (data == BUILT_UP).astype("float32"), transform


def built_fraction(units, built: np.ndarray, transform, step_m: float = 30.0) -> pd.Series:
    share = focal_mean(built, _odd(400 / (abs(transform.a) * DEG_M)))
    values = {unit_id: _agg(np.nanmean, sample(share, transform, lons, lats))
              for unit_id, lons, lats in _unit_lonlat(units, step_m)}
    return pd.Series(values, name="built_frac_200")
```

```python
from __future__ import annotations

import sys
from pathlib import Path

import geopandas as gpd
import pandas as pd

from floodrisk import contracts
from floodrisk.config import CITIES, units_path
from floodrisk.data import terrain
from floodrisk.data.cells import nearest_cell, rain_cells
from floodrisk.data.units import build_units

WATER_TAGS = {"waterway": ["river", "canal", "stream", "drain"], "natural": ["water"]}


def download_water(city: str) -> gpd.GeoDataFrame:
    import osmnx as ox

    return ox.features_from_bbox(CITIES[city].bbox, tags=WATER_TAGS)[["geometry"]].reset_index(drop=True)


def water_distance(units: gpd.GeoDataFrame, water: gpd.GeoDataFrame) -> pd.Series:
    crs = units.estimate_utm_crs()
    joined = gpd.sjoin_nearest(
        units[["unit_id", "geometry"]].to_crs(crs), water[["geometry"]].to_crs(crs),
        how="left", distance_col="dist_water_m",
    )
    return joined.groupby("unit_id")["dist_water_m"].min()


def assemble(units: gpd.GeoDataFrame, terrain_df: pd.DataFrame, built: pd.Series, water: pd.Series, cell: pd.Series) -> gpd.GeoDataFrame:
    out = units.merge(terrain_df, on="unit_id", how="left")
    out["built_frac_200"] = out["unit_id"].map(built).astype(float)
    out["dist_water_m"] = out["unit_id"].map(water).astype(float)
    out["rain_cell"] = out["unit_id"].map(cell).astype(int)
    contracts.validate_units(out)
    return out


def build(city: str, dem_dir) -> Path:
    bbox = CITIES[city].bbox
    units = build_units(city)
    dem, dem_transform = terrain.load_dem(dem_dir, bbox)
    terrain_df = terrain.terrain_features(units, dem, dem_transform)
    built, built_transform = terrain.load_built(bbox)
    built_share = terrain.built_fraction(units, built, built_transform)
    water = water_distance(units, download_water(city))
    centre = units.geometry.representative_point()
    cell = pd.Series(nearest_cell(centre.x.to_numpy(), centre.y.to_numpy(), rain_cells(city)), index=units["unit_id"].to_numpy())
    out = assemble(units, terrain_df, built_share, water, cell)
    path = units_path(city)
    path.parent.mkdir(parents=True, exist_ok=True)
    out.to_parquet(path)
    return path


if __name__ == "__main__":
    print(build(sys.argv[1], Path(sys.argv[2])))
```

```python
def match_units(obs: pd.DataFrame, units: gpd.GeoDataFrame, near_m: float = 100.0, name_m: float = 300.0) -> pd.Series:
    crs = units.estimate_utm_crs()
    u = units[["unit_id", "name", "geometry"]].to_crs(crs).reset_index(drop=True)
    keys = u["name"].map(norm_name)
    points = gpd.GeoSeries(gpd.points_from_xy(obs["lon"], obs["lat"]), crs=4326).to_crs(crs)
    out = pd.Series([None] * len(obs), index=obs.index, dtype=object)
    for idx, point, street in zip(obs.index, points, obs["street_name"]):
        candidates = u.sindex.query(point.buffer(name_m))
        if len(candidates) == 0:
            continue
        dist = u.geometry.iloc[candidates].distance(point)
        key = norm_name(street) if isinstance(street, str) else ""
        same_name = dist[(keys.iloc[candidates] == key) & (dist <= name_m)] if key else dist.iloc[0:0]
        pick = same_name if len(same_name) else dist[dist <= near_m]
        if len(pick):
            out.loc[idx] = u["unit_id"].loc[pick.idxmin()]
    return out
```

```python
from __future__ import annotations

import json
import logging
import os
from pathlib import Path

import joblib
import pandas as pd

from floodrisk import contracts
from floodrisk.config import (CITIES, TZ, rain_cells_path, risk_path, scores_path, thresholds_path,
                              tide_coef_path, trigger_path, units_path)

log = logging.getLogger("floodrisk.hourly")


def run_city(city: str, now: pd.Timestamp | None = None, fetch=None, scorer=None) -> Path:
    if fetch is None:
        from floodrisk.data.rain import fetch_forecast as fetch
    if scorer is None:
        from floodrisk.model.scoring import score_units as scorer
    now = now if now is not None else pd.Timestamp.now(tz=TZ).tz_localize(None).floor("h")

    cells = pd.read_parquet(rain_cells_path(city))
    units = pd.read_parquet(units_path(city), columns=["unit_id", "rain_cell"])
    scores = pd.read_parquet(scores_path(city)).merge(units, on="unit_id")
    trigger = json.loads(trigger_path(city).read_text(encoding="utf-8"))
    thresholds = json.loads(thresholds_path().read_text(encoding="utf-8"))

    tide_fn = None
    if CITIES[city].has_tide and trigger.get("tide") and tide_coef_path().exists():
        from floodrisk.data.tide import window_max

        coef = joblib.load(tide_coef_path())
        tide_fn = lambda at: window_max(coef, at)

    rain = fetch(city, cells)
    risk = scorer(scores, rain, trigger, thresholds, now, tide_fn=tide_fn)
    risk["computed_at"] = now
    contracts.validate(risk, contracts.RISK, "risk")

    path = risk_path(city)
    tmp = path.with_suffix(".parquet.tmp")
    risk.to_parquet(tmp)
    os.replace(tmp, path)
    return path


def run_all() -> list[str]:
    done = []
    for city in CITIES:
        if not scores_path(city).exists():
            continue
        try:
            run_city(city)
            done.append(city)
        except Exception:
            log.exception("Tác vụ giờ lỗi ở %s; giữ nguyên lớp nguy cơ cũ", city)
    return done


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    print(run_all())
```
